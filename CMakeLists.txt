cmake_minimum_required(VERSION 3.16)
project(raftkv VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(RAFTKV_BUILD_TESTS "Build tests" ON)
option(RAFTKV_BUILD_GRPC "Build the gRPC server, client and benchmark" ON)
option(RAFTKV_WERROR "Warnings as errors" OFF)
set(RAFTKV_SANITIZE "" CACHE STRING "Sanitizers: 'address;undefined' or 'thread'")
find_package(Threads REQUIRED)

function(raftkv_configure tgt)
  target_compile_options(${tgt} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
  if(RAFTKV_WERROR)
    target_compile_options(${tgt} PRIVATE -Werror)
  endif()
  if(RAFTKV_SANITIZE)
    string(REPLACE ";" "," _san "${RAFTKV_SANITIZE}")
    target_compile_options(${tgt} PRIVATE -fsanitize=${_san} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(${tgt} PRIVATE -fsanitize=${_san})
  endif()
endfunction()

add_library(raftkv_core STATIC src/raft.cpp src/wal.cpp src/posix_fs.cpp src/replica.cpp src/lincheck.cpp src/sim.cpp)
target_include_directories(raftkv_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
raftkv_configure(raftkv_core)

if(RAFTKV_BUILD_TESTS)
  enable_testing()
  file(GLOB TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
  add_executable(raftkv_tests ${TEST_SOURCES})
  target_link_libraries(raftkv_tests PRIVATE raftkv_core Threads::Threads)
  target_include_directories(raftkv_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  raftkv_configure(raftkv_tests)
  target_compile_options(raftkv_tests PRIVATE -O1 -g -UNDEBUG)
  add_test(NAME raftkv_tests COMMAND raftkv_tests)
endif()

add_executable(raftkv_sim src/sim_main.cpp)
target_link_libraries(raftkv_sim PRIVATE raftkv_core Threads::Threads)
raftkv_configure(raftkv_sim)
target_compile_options(raftkv_sim PRIVATE -O2)

if(RAFTKV_BUILD_GRPC)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GRPCPP REQUIRED IMPORTED_TARGET grpc++)
  pkg_check_modules(PROTOBUF REQUIRED IMPORTED_TARGET protobuf)
  find_program(PROTOC protoc REQUIRED)
  find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin REQUIRED)
  set(GEN ${CMAKE_CURRENT_BINARY_DIR}/gen)
  file(MAKE_DIRECTORY ${GEN})
  add_custom_command(
    OUTPUT ${GEN}/raftkv.pb.cc ${GEN}/raftkv.pb.h ${GEN}/raftkv.grpc.pb.cc ${GEN}/raftkv.grpc.pb.h
    COMMAND ${PROTOC} --cpp_out=${GEN} --grpc_out=${GEN} --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN}
            -I ${CMAKE_CURRENT_SOURCE_DIR}/proto ${CMAKE_CURRENT_SOURCE_DIR}/proto/raftkv.proto
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/proto/raftkv.proto)
  add_library(raftkv_rpc STATIC ${GEN}/raftkv.pb.cc ${GEN}/raftkv.grpc.pb.cc src/node_server.cpp src/kv_client.cpp)
  set_source_files_properties(${GEN}/raftkv.pb.cc ${GEN}/raftkv.grpc.pb.cc PROPERTIES COMPILE_OPTIONS "-w")
  target_include_directories(raftkv_rpc PUBLIC ${GEN} ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(raftkv_rpc PUBLIC raftkv_core PkgConfig::GRPCPP PkgConfig::PROTOBUF Threads::Threads)
  raftkv_configure(raftkv_rpc)
  foreach(t raftkv_server raftkv_bench raftkv_ctl)
    string(REPLACE "raftkv_" "" short ${t})
    if(short STREQUAL "server")
      set(src src/server_main.cpp)
    elseif(short STREQUAL "bench")
      set(src src/bench_main.cpp)
    else()
      set(src src/ctl_main.cpp)
    endif()
    add_executable(${t} ${src})
    target_link_libraries(${t} PRIVATE raftkv_rpc)
    raftkv_configure(${t})
    target_compile_options(${t} PRIVATE -O2)
  endforeach()
endif()
